<#

    LSP Offboarding Automator
    Automates Standard and Immediate offboarding procedures for Legacy Service Partners.

# -------------------------------------------------------------------------
# FUNCTIONS & EXECUTION LOGIC
# -------------------------------------------------------------------------
function Write-GuiLog ($Message) {
    $Timestamp = Get-Date -Format "HH:mm:ss"
    $txtLog.AppendText("[$Timestamp] $Message`r`n")
    $form.Refresh()
}

$btnSubmit.Add_Click({
    $UPN = $txtUPN.Text.Trim()
    $Type = $cmbType.SelectedItem.ToString()
    $ContactEmail = $txtContact.Text.Trim()

    if ([string]::IsNullOrWhiteSpace($UPN)) {
        [System.Windows.Forms.MessageBox]::Show("Please enter a User UPN.", "Error", [System.Windows.Forms.MessageBoxButtons]::OK, [System.Windows.Forms.MessageBoxIcon]::Error)
        return
    }

    $txtLog.Clear()
    Write-GuiLog "Starting $Type for $UPN..."

    try {
        # 1. Connect to Microsoft Graph & Exchange
        Write-GuiLog "Connecting to Microsoft Graph and Exchange Online..."
        Connect-MgGraph -Scopes "User.ReadWrite.All", "Directory.ReadWrite.All", "GroupMember.ReadWrite.All" -NoWelcome
        Connect-ExchangeOnline -ShowProgress $false -ShowBanner $false

        # 2. Retrieve the User
        $User = Get-MgUser -UserId $UPN -ErrorAction Stop
        $OldName = $User.DisplayName

        # 3. Step 01: Disable O365 Account (Runs for both Immediate and Standard)
        Write-GuiLog "Blocking sign-in and resetting password..."
        $RandomPassword = "LSP-" + ([guid]::NewGuid().ToString().Substring(0,8)) + "!A1"
        $PasswordProfile = @{
            Password = $RandomPassword
            ForceChangePasswordNextSignIn = $false
        }
        Update-MgUser -UserId $User.Id -AccountEnabled:$false -PasswordProfile $PasswordProfile
        
        Write-GuiLog "Revoking all active sign-in sessions..."
        Revoke-MgUserSignInSession -UserId $User.Id | Out-Null

        # 4. Halt if Immediate Offboarding
        if ($Type -like "*Immediate*") {
            Write-GuiLog "========================================"
            Write-GuiLog "IMMEDIATE OFFBOARDING LOCKDOWN COMPLETE."
            Write-GuiLog "Account Disabled: Yes"
            Write-GuiLog "Sessions Revoked: Yes"
            Write-GuiLog "Password Reset: Yes"
            Write-GuiLog "========================================"
            Write-GuiLog "ACTION REQUIRED: Please proceed with ServiceTitan & Adobe offboarding manually, then close the P1 ticket."
            return
        }

        # 5. Step 02: Modify O365 Account (Standard Only)
        Write-GuiLog "Hiding user from Global Address List (GAL)..."
        Set-Mailbox -Identity $UPN -HiddenFromAddressListsEnabled $true

        Write-GuiLog "Prepending 'ZZ -' to Display Name..."
        $NewName = "ZZ - $OldName"
        Update-MgUser -UserId $User.Id -DisplayName $NewName

        # Conditional Shared Mailbox
        if ($chkShared.Checked) {
            Write-GuiLog "Converting mailbox to Shared..."
            Set-Mailbox -Identity $UPN -Type Shared
            $SharedStatus = "Yes"
        } else {
            Write-GuiLog "Skipping Shared Mailbox conversion per user selection..."
            $SharedStatus = "Skipped"
        }

        # Conditional Auto-Reply
        if ($chkAutoReply.Checked) {
            Write-GuiLog "Configuring Auto-Reply..."
            $AutoReplyText = "This mailbox is no longer being monitored. Please contact $ContactEmail for assistance."
            Set-MailboxAutoReplyConfiguration -Identity $UPN -AutoReplyState Enabled -InternalMessage $AutoReplyText -ExternalMessage $AutoReplyText
            $ReplyStatus = "Yes"
        } else {
            Write-GuiLog "Skipping Auto-Reply configuration per user selection..."
            $ReplyStatus = "Skipped"
        }

        Write-GuiLog "Applying Dynamic Group Exclusion Custom Attributes..."
        Set-Mailbox -Identity $UPN -CustomAttribute1 "Exclude All Legacy" -CustomAttribute2 "Exclude All Company" -CustomAttribute3 "Exclude All Users" -CustomAttribute15 "Exclude Legacy Service Partners Group"

        Write-GuiLog "Stripping Group Memberships (This automatically removes Group-Based Licenses like AZ_E1/AZ_E3)..."
        $Groups = Get-MgUserMemberOf -UserId $User.Id
        foreach ($Group in $Groups) {
            if ($Group.AdditionalProperties["@odata.type"] -eq "#microsoft.graph.group") {
                Remove-MgGroupMemberByRef -GroupId $Group.Id -DirectoryObjectId $User.Id -ErrorAction SilentlyContinue
            }
        }

        Write-GuiLog "Removing direct O365 licenses..."
        $Licenses = Get-MgUserLicenseDetail -UserId $User.Id
        foreach ($License in $Licenses) {
            Set-MgUserLicense -UserId $User.Id -RemoveLicenses @($License.SkuId) -AddLicenses @{} -ErrorAction SilentlyContinue
        }

        Write-GuiLog "========================================"
        Write-GuiLog "STANDARD OFFBOARDING COMPLETE."
        Write-GuiLog "User: $UPN"
        Write-GuiLog "New Name: $NewName"
        Write-GuiLog "- Locked out & Sessions Revoked: Yes"
        Write-GuiLog "- Hidden from GAL: Yes"
        Write-GuiLog "- Converted to Shared: $SharedStatus"
        Write-GuiLog "- Auto-Reply Configured: $ReplyStatus"
        Write-GuiLog "- Custom Attributes (1,2,3,15) Applied: Yes"
        Write-GuiLog "- Groups & Licenses Removed: Yes"
        Write-GuiLog "========================================"
        Write-GuiLog "REMINDER: Verify Admin Roles and offboard external apps (ServiceTitan/Adobe) if applicable."

    } catch {
        Write-GuiLog "ERROR: $($_.Exception.Message)"
    }
})

$form.ShowDialog() | Out-Null
